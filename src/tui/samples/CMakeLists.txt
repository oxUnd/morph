cmake_minimum_required(VERSION 3.20)

# Modern linkers can resolve static archives independent of their command-line
# position. Let CMake remove duplicate dependencies using that capability.
if(POLICY CMP0156)
    cmake_policy(SET CMP0156 NEW)
endif()
if(POLICY CMP0179)
    cmake_policy(SET CMP0179 NEW)
endif()
if(CMAKE_VERSION VERSION_GREATER_EQUAL 3.31)
    set(CMAKE_LINK_LIBRARIES_STRATEGY REORDER_FREELY)
endif()

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
	project(morph-tui-samples LANGUAGES C)
	set(CMAKE_C_STANDARD 11)
	set(CMAKE_C_STANDARD_REQUIRED ON)
	set(MORPH_TUI_SAMPLES_STANDALONE ON)
	find_package(Threads REQUIRED)
	find_package(PkgConfig REQUIRED)
	get_filename_component(MORPH_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)

	set(NB_BUILD_SHARED OFF CACHE BOOL "" FORCE)
	set(NB_BUILD_TESTS OFF CACHE BOOL "" FORCE)
	set(NB_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
	set(NB_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
	set(NB_BUILD_FUZZERS OFF CACHE BOOL "" FORCE)
	add_subdirectory("${MORPH_SOURCE_ROOT}/vendor/nothing" "${CMAKE_CURRENT_BINARY_DIR}/nbase")
	add_subdirectory("${MORPH_SOURCE_ROOT}/src/media" "${CMAKE_CURRENT_BINARY_DIR}/media")
	add_subdirectory("${MORPH_SOURCE_ROOT}/src/tui" "${CMAKE_CURRENT_BINARY_DIR}/tui")
endif()

add_subdirectory(minimal)
add_subdirectory(controls)
add_subdirectory(customization)
