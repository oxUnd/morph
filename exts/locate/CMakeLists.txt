cmake_minimum_required(VERSION 3.20)
project(morph-locate LANGUAGES C)

set(NB_BUILD_SHARED OFF CACHE BOOL "Link static nbase" FORCE)
set(NB_BUILD_TESTS OFF CACHE BOOL "Build nbase tests" FORCE)
set(NB_BUILD_EXAMPLES OFF CACHE BOOL "Build nbase examples" FORCE)
add_subdirectory(../../vendor/nothing nbase)

if(APPLE)
	enable_language(OBJC)
	set_source_files_properties(locate.c PROPERTIES LANGUAGE OBJC)
endif()
add_executable(locate locate.c)
target_link_libraries(locate PRIVATE nbase::nbase m)
target_compile_options(locate PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Wconversion)
if(APPLE)
	target_link_libraries(locate PRIVATE "-framework Foundation" "-framework CoreLocation")
else()
	find_package(PkgConfig REQUIRED)
	pkg_check_modules(SYSTEMD REQUIRED IMPORTED_TARGET libsystemd)
	target_link_libraries(locate PRIVATE PkgConfig::SYSTEMD)
endif()
